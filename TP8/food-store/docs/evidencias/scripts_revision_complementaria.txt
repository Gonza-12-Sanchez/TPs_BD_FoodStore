--Creacion de una base de datos


-- Tipos enumerados
CREATE TYPE rol           AS ENUM ('ADMIN','USUARIO');
CREATE TYPE estado_pedido AS ENUM ('PENDIENTE','CONFIRMADO',
                                   'TERMINADO','CANCELADO');
CREATE TYPE forma_pago    AS ENUM ('TARJETA','TRANSFERENCIA','EFECTIVO');
 
-- Creacion de tablas con distintas validaciones
CREATE TABLE categoria (
    id          BIGINT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    nombre      VARCHAR(80)  NOT NULL UNIQUE,
    descripcion VARCHAR(255),
    eliminado   BOOLEAN      NOT NULL DEFAULT FALSE,
    created_at  TIMESTAMPTZ  NOT NULL DEFAULT now()
);
 
CREATE TABLE producto (
    id           BIGINT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    nombre       VARCHAR(120) NOT NULL,
    precio       NUMERIC(10,2) NOT NULL CHECK (precio >= 0),
    descripcion  VARCHAR(255),
    stock        INTEGER      NOT NULL DEFAULT 0 CHECK (stock >= 0),
    imagen       VARCHAR(255),
    disponible   BOOLEAN      NOT NULL DEFAULT TRUE,
    categoria_id BIGINT       NOT NULL REFERENCES categoria(id),
    eliminado    BOOLEAN      NOT NULL DEFAULT FALSE,
    created_at   TIMESTAMPTZ  NOT NULL DEFAULT now()
);
 
CREATE TABLE usuario (
    id          BIGINT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    nombre      VARCHAR(80)  NOT NULL,
    apellido    VARCHAR(80)  NOT NULL,
    mail        VARCHAR(120) NOT NULL UNIQUE,
    celular     VARCHAR(30),
    contrasena  VARCHAR(255) NOT NULL,
    rol         rol          NOT NULL DEFAULT 'USUARIO',
    eliminado   BOOLEAN      NOT NULL DEFAULT FALSE,
    created_at  TIMESTAMPTZ  NOT NULL DEFAULT now()
);
 
CREATE TABLE pedido (
    id         BIGINT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    fecha      DATE          NOT NULL DEFAULT CURRENT_DATE,
    estado     estado_pedido NOT NULL DEFAULT 'PENDIENTE',
    total      NUMERIC(12,2) NOT NULL DEFAULT 0 CHECK (total >= 0),
    forma_pago forma_pago    NOT NULL,
    usuario_id BIGINT        NOT NULL REFERENCES usuario(id),
    eliminado  BOOLEAN       NOT NULL DEFAULT FALSE,
    created_at TIMESTAMPTZ   NOT NULL DEFAULT now()
);
    
CREATE TABLE detalle_pedido (
    id          BIGINT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    cantidad       INTEGER       NOT NULL CHECK (cantidad > 0),
    precio_unitario NUMERIC(10,2) NOT NULL CHECK (precio_unitario >= 0),
    subtotal       NUMERIC(12,2) NOT NULL CHECK (subtotal >= 0),
    pedido_id      BIGINT        NOT NULL REFERENCES pedido(id)
                                 ON DELETE RESTRICT,
    producto_id    BIGINT        NOT NULL REFERENCES producto(id),
    eliminado      BOOLEAN       NOT NULL DEFAULT FALSE,
    created_at     TIMESTAMPTZ   NOT NULL DEFAULT now(),
    UNIQUE (pedido_id, producto_id)
);

-- Creacion de indices
create index idx_producto_categoria on producto(categoria_id);
create index idx_pedido_usuario on pedido(usuario_id);
create index idx_detalle_pedido on detalle_pedido(pedido_id);
create index idx_pedido_estado_fecha on pedido(estado,fecha);
create index idx_producto_desc on producto using GIN (to_tsvector('spanish', descripcion));
create index idx_detalle_producto on detalle_pedido(producto_id);
create index idx_producto_nombre_vig on producto(nombre) where eliminado = FALSE;

def pg(sql=None, file=None, db=None):
    cmd = [args.psql, '-X', '-w', '-h', args.host, '-p', args.port, '-U', args.user,
           '-v','ON_ERROR_STOP=1','-A','-t','-d',db or args.database]
    cmd += ['-f',str(file)] if file else ['-c',sql]
    env = dict(os.environ, PGCLIENTENCODING='UTF8', PGCONNECT_TIMEOUT='5')
    p = subprocess.run(cmd, capture_output=True, encoding='utf-8', errors='replace', env=env, timeout=60)
    if p.returncode:
        raise RuntimeError(p.stderr)
    return p.stdout.strip()

def source(relative):
    return (ROOT/relative).read_text(encoding='utf-8-sig')

def statements(text):
    buf = ''
    for line in text.splitlines(True):
        buf += line
        if sqlite3.complete_statement(buf):
            yield buf
            buf = ''
    if buf.strip() and not all(x.strip().startswith('--') or not x.strip() for x in buf.splitlines()):
        raise ValueError('SQL incompleto')

def run_sql(text):
    # execute por sentencia permite conservar una transacción abierta en importación.
    for stmt in statements(text):
        con.execute(stmt)

def export_import(label):
    csv_text = pg('COPY (SELECT * FROM v_snapshot_movil ORDER BY id) TO STDOUT WITH CSV HEADER;') + '\n'
    file = out/('snapshot_'+label+'.csv')
    file.write_text(csv_text, encoding='utf-8', newline='')
    rows = list(csv.DictReader(io.StringIO(csv_text)))
    if not rows:
        raise ValueError('El laboratorio requiere una instantánea no vacía.')
    if len({r['fecha_copia'] for r in rows}) != 1:
        raise ValueError('El CSV debe corresponder a una única instantánea.')
    with con:
        con.execute('DELETE FROM snapshot_servidor')
        con.executemany('INSERT INTO snapshot_servidor VALUES (?,?,?,?,?,?,?)',
            [(int(r['id']),r['nombre'],int(r['precio_centavos']),int(r['stock']),r['disponible'],r['eliminado'],r['fecha_copia']) for r in rows])
        run_sql(source('movil/cargar_cache.sql'))
    record(f'{label}: exportación real PostgreSQL -> CSV -> SQLite; fecha={rows[0]["fecha_copia"]}; filas={len(rows)}')
    return file

"""Arranca una instancia descartable en localhost, prueba y la detiene."""
import argparse
import os
from pathlib import Path
import subprocess
import sys
import tempfile

sys.stdout.reconfigure(encoding='utf-8',errors='replace')
p=argparse.ArgumentParser(description=__doc__)
p.add_argument('--pg-bin',default=r'E:\PostgreSQL\bin')
p.add_argument('--port',default='55439')
a=p.parse_args()
root=Path(__file__).resolve().parent
bin=Path(a.pg_bin)
if not (bin/'initdb.exe').is_file():
    p.error('Indicá --pg-bin con la carpeta bin de PostgreSQL.')
lab=Path(tempfile.mkdtemp(prefix='foodstore_tp8_'))
data=lab/'pg'
env=dict(os.environ,PYTHONIOENCODING='utf-8')
def run(cmd,name,utf8=False):
    # Archivos, no PIPE: el servidor puede heredar handles en Windows.
    stdout=lab/(name+'.stdout.log')
    stderr=lab/(name+'.stderr.log')
    with stdout.open('wb') as so, stderr.open('wb') as se:
        result=subprocess.run([str(x) for x in cmd],stdout=so,stderr=se,env=env)
    encoding='utf-8' if utf8 else 'cp1252'
    print(stdout.read_bytes().decode(encoding,errors='replace'),flush=True)
    if result.returncode:
        raise RuntimeError(name+': '+stderr.read_bytes().decode(encoding,errors='replace'))
started=False
try:
    run([bin/'initdb.exe','-D',data,'-U','postgres','-A','trust','--encoding=UTF8','--no-locale'],'initdb')
    run([bin/'pg_ctl.exe','-D',data,'-l',lab/'server.log','-o','-p '+a.port+' -h 127.0.0.1','-w','start'],'inicio')
    started=True
    run([sys.executable,root/'ejecutar.py','--psql',bin/'psql.exe','--port',a.port],'pruebas',True)
finally:
    if started:
        run([bin/'pg_ctl.exe','-D',data,'-m','fast','-w','stop'],'cierre')
    print('Laboratorio conservado en '+str(lab))
